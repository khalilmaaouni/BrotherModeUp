#!/usr/bin/env python3
"""Windows-shaped probes, executed on macOS against the SHIPPED functions.

sys.platform is forced to 'win32' around each call, so the branch that
really runs on Windows is the branch measured here. This gives exact
LEXICAL Windows semantics (the helper is pure Python); it does NOT
execute cmd.exe, and every claim about what cmd.exe then does with the
string is labelled as reasoning, not measurement.
"""
import importlib.util
import io
import ntpath
import os
import shlex
import subprocess
import sys
import tempfile

HERE = "/Users/khalil.maaouni/Documents/BrotherModeUp/tools"


def _load(name):
    spec = importlib.util.spec_from_file_location(
        name, os.path.join(HERE, name + ".py"))
    mod = importlib.util.module_from_spec(spec)
    sys.modules[name] = mod
    spec.loader.exec_module(mod)
    return mod


bs = _load("bm_store")


class _AsWindows(object):
    def __enter__(self):
        self.saved = sys.platform
        sys.platform = "win32"

    def __exit__(self, *exc):
        sys.platform = self.saved
        return False


def q_win(path):
    with _AsWindows():
        return bs._quote_path_for_local_shell(path)


def q_posix(path):
    return bs._quote_path_for_local_shell(path)


def head(title):
    print("")
    print("=" * 72)
    print(title)
    print("=" * 72)


WIN_ROOTS = [
    r"C:\Users\RUNNER~1\AppData\Local\Temp\tmp8x1\P",
    r"C:\Users\runneradmin\My Projects\tmp8x1\P",   # a space
    r"C:\R&D\tmp8x1\P",                             # a cmd metacharacter

                                                shlex.quote(root))
    new = "git --git-dir=%s --work-tree=%s " % (q_win(marker), q_win(root))
    print("")
    print("root: %s" % root)
    print("  OLD (shlex.quote)          : %srestore -- a.py" % old)
    print("  NEW (_quote_path_for_local_shell): %srestore -- a.py" % new)
    print("  OLD contains a single quote: %s" % ("'" in old))
    print("  NEW contains a single quote: %s" % ("'" in new))
